# SPIKE (Phase 0, 0.0.8). Everything logs to game.log with the prefix "AHW SPIKE".
# In effects with parameters, "[" in a log string must be written "\\[" (run 1).
# A $PARAM$ next to "\\[" comes out mangled (run 2), so parameters go last in each log line.

# One ring node $NAME$, $R$ from the scope (no scope: the galaxy's origin?), direction $ANGLE$.
ahw_spike_spawn_ring_node = {
	spawn_system = {
		min_distance = $R$
		max_distance = $R$
		max_jumps = 0
		min_orientation_angle = $ANGLE$
		max_orientation_angle = $ANGLE$
		initializer = ahw_spike_ring_init
		hyperlane = no
		is_discovered = yes
		effect = {
			set_star_flag = ahw_ring
			set_star_flag = ahw_unclaimable
			set_name = $NAME$
			save_global_event_target_as = $NAME$
		}
	}
}

# this = system
ahw_spike_log_system = {
	export_trigger_value_to_variable = {
		trigger = distance_to_core_percent
		variable = ahw_core_pct
	}
	log = "AHW SPIKE \\[This.GetName]: distance_to_core_percent (a fraction) = \\[This.ahw_core_pct], exits = \\[This.ahw_exits], pair = \\[This.ahw_pair] -- $WHAT$"
	if = { limit = { has_star_flag = ahw_broken } log = "AHW SPIKE   \\[This.GetName]: BROKEN" }
	if = { limit = { has_star_flag = ahw_studied } log = "AHW SPIKE   \\[This.GetName]: studied" }
	if = {
		limit = { exists = owner }
		log = "AHW SPIKE   \\[This.GetName]: owned by \\[This.Owner.GetName]"
	}
	every_bypass_in_system = {
		if = {
			limit = { has_lock = yes }
			log = "AHW SPIKE   a bypass here is LOCKED"
		}
	}
}

# How empty is the core? distance_to_core_percent is a fraction 0-1 (run 2).
ahw_spike_core_band = {
	if = {
		limit = { any_system = { distance_to_core_percent < $FRACTION$ } }
		log = "AHW SPIKE   some systems are closer to the centre than $FRACTION$"
	}
	else = {
		log = "AHW SPIKE   NO system is closer to the centre than $FRACTION$"
	}
}

# this = country (the player)
ahw_spike_report = {
	log = "AHW SPIKE ===== report: $WHEN$ ====="
	if = {
		limit = { has_technology = ahw_tech_spike }
		log = "AHW SPIKE \\[This.GetName] has ahw_tech_spike"
	}
	else = {
		log = "AHW SPIKE \\[This.GetName] does NOT have ahw_tech_spike"
	}
	every_system = {
		limit = { has_star_flag = ahw_ring }
		ahw_spike_log_system = { WHAT = ring_node }
	}
	every_system = {
		limit = { has_star_flag = ahw_portal }
		ahw_spike_log_system = { WHAT = portal }
	}
	every_system = {
		limit = { has_star_flag = ahw_core }
		ahw_spike_log_system = { WHAT = centre_probe }
	}
	if = {
		limit = { exists = event_target:ahw_player_portal }
		event_target:ahw_player_portal = { ahw_spike_log_system = { WHAT = OUR_PORTAL } }
	}
}

# this = system: its distance_to_core_percent into ahw_d.
ahw_spike_measure = {
	export_trigger_value_to_variable = {
		trigger = distance_to_core_percent
		variable = ahw_d
	}
}

# One leapfrog step towards the centre: probe T moves $S$ corewards of probe A; if T is now closer
# to the centre, the two swap roles (ahw_probe_a is always the best so far).
ahw_spike_probe_step = {
	event_target:ahw_probe_t = {
		move_system = {
			target = event_target:ahw_probe_a
			min_distance >= $S$
			max_distance <= $S$
			max_jumps = 0
			direction = corewards
			hyperlane = no
		}
		ahw_spike_measure = yes
	}
	if = {
		limit = {
			event_target:ahw_probe_t = { check_variable = { which = ahw_d value < event_target:ahw_probe_a.ahw_d } }
		}
		event_target:ahw_probe_t = { save_global_event_target_as = ahw_probe_swap }
		event_target:ahw_probe_a = { save_global_event_target_as = ahw_probe_t }
		event_target:ahw_probe_swap = { save_global_event_target_as = ahw_probe_a }
	}
}

# this = a portal: locks both ends of its pair (the gate here, the exit in the ring) for $COUNTRY$,
# $DAYS$ days. A lock only stops ships entering that end (run 18), so both ends are needed.
ahw_spike_lock_pair = {
	every_bypass_in_system = {
		lock_bypass = { country = $COUNTRY$ duration = $DAYS$ }
	}
	ahw_spike_lock_ring_end = { COUNTRY = $COUNTRY$ DAYS = $DAYS$ }
}

# this = the gate megastructure (overclock mode Sealed): both ends locked for its owner, for good
# (36000 days; renewed by the monthly check while sealed).
ahw_spike_seal_gate = {
	if = {
		limit = { exists = owner }
		owner = { save_event_target_as = ahw_sealer }
		solar_system = {
			set_star_flag = ahw_sealed
			ahw_spike_lock_pair = { COUNTRY = event_target:ahw_sealer DAYS = 36000 }
			log = "AHW SPIKE gate in \\[This.GetName] SEALED for \\[ahw_sealer.GetName]"
		}
	}
}

# this = the gate megastructure (overclock mode Open): if it was sealed, both ends re-locked for the
# owner for 1 day - they open tomorrow (no unlock effect exists). Also runs when the mode is first
# set at creation, hence the flag check.
ahw_spike_unseal_gate = {
	if = {
		limit = {
			exists = owner
			solar_system = { has_star_flag = ahw_sealed }
		}
		owner = { save_event_target_as = ahw_sealer }
		solar_system = {
			remove_star_flag = ahw_sealed
			ahw_spike_lock_pair = { COUNTRY = event_target:ahw_sealer DAYS = 1 }
			log = "AHW SPIKE gate in \\[This.GetName] opened by \\[ahw_sealer.GetName] (both ends free tomorrow)"
		}
	}
}
