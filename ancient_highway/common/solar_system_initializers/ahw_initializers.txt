# a ring system. Its centre: our black hole (common/star_classes/
# ahw_star_classes.txt), at 20 % of vanilla's size (vanilla 20-35); ahw_pick_centres
# turns two in three into a pulsar or a toxoid star after the build. Run 24: an initializer whose
# class is a random list of our classes left the systems EMPTY (no star, no asteroid; "Failed to
# find star class or valid random_list by key") - likely because they all have spawn_odds = 0.
ahw_ring_init = {
	name = NAME_ahw_ring
	class = "ahw_sc_black_hole"
	prevent_anomalies = yes

	flags = { ahw_ring ahw_unclaimable suppress_archaeological_sites }

	planet = {
		class = "pc_black_hole"
		orbit_distance = 0
		orbit_angle = 1
		size = 6
		has_ring = no
	}

	# 0.0.16 (user, run 15: the ring is too fast - make its systems bigger, so the crossing also
	# speeds up with better ships). Hyperlanes arrive at max(SYSTEM_MIN_INNER_RADIUS = 150, the
	# outermost planet's orbit + SYSTEM_INNER_RADIUS_OFFSET = 30) (common/defines/00_defines.txt):
	# a body at orbit 270 puts the arrivals at 300, double the black hole's 150. Orbits in an
	# initializer add up (run 3); the centre is at 0. Q: does an asteroid count, and how much
	# slower is a node then (run 15: ~70 days each)?
	planet = {
		class = "pc_asteroid"
		orbit_distance = 270
		orbit_angle = 90
		size = 1
		has_ring = no
	}
}
