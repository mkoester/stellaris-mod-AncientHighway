# SPIKE (Phase 0, 0.0.8). this = system: may the portal hang off this system? Not the capital
# (event_target:ahw_cap_tmp), not next to it, not artificial, not owned (game start: only home
# systems are). Used with ahw_spike_within_3_jumps_of_capital: 2 or 3 jumps out.
ahw_spike_portal_anchor_ok = {
	NOT = { is_same_value = event_target:ahw_cap_tmp }
	NOT = { has_hyperlane_to = event_target:ahw_cap_tmp }
	NOT = { has_star_flag = ahw_unclaimable }
	NOT = { exists = owner }
	# Run 12: two empires shared the dead end Tiblic; the second portal's lane arrived beside the
	# first's (margin -0.55). One portal per anchor.
	NOT = { any_neighbor_system = { has_star_flag = ahw_portal } }
}

# Run 11: the random walk saw one path only and missed 3-jump candidates; ahw_spike.10 now
# searches every system. this = system: 1 to 3 jumps from the capital (with
# ahw_spike_portal_anchor_ok, which rules out the capital and its neighbours: 2 to 3 jumps).
ahw_spike_within_3_jumps_of_capital = {
	OR = {
		has_hyperlane_to = event_target:ahw_cap_tmp
		any_neighbor_system = { has_hyperlane_to = event_target:ahw_cap_tmp }
		any_neighbor_system = { any_neighbor_system = { has_hyperlane_to = event_target:ahw_cap_tmp } }
	}
}

# this = system: already holds a gateway, wormhole or other bypass (user: de-prioritise these).
ahw_spike_has_bypass = {
	any_bypass_in_system = { always = yes }
}

# 0.0.24 (user, 2026-11-05: "the whole idea is that empires can still reach their allies - every
# type of friend must still be allowed through (esp. empires fighting on the same side of a war)").
# this = country: the sealer of pair $N$ (event_target:ahw_sealer_$N$) or a friend of it. A pair
# sealed against everyone (ahw_spike.25) has no sealer, so nobody passes.
ahw_spike_friend_of_sealer = {
	exists = event_target:ahw_sealer_$N$
	OR = {
		is_same_value = event_target:ahw_sealer_$N$
		# one federation
		is_in_federation_with = event_target:ahw_sealer_$N$
		# overlord and subject, either way, and two subjects of one overlord
		is_overlord_to = event_target:ahw_sealer_$N$
		event_target:ahw_sealer_$N$ = { is_overlord_to = prev }
		AND = {
			is_subject = yes
			event_target:ahw_sealer_$N$ = { is_subject = yes }
			overlord = { is_overlord_to = event_target:ahw_sealer_$N$ }
		}
		# pacts and guarantees, either way
		has_defensive_pact = event_target:ahw_sealer_$N$
		has_non_aggression_pact = event_target:ahw_sealer_$N$
		has_commercial_pact = event_target:ahw_sealer_$N$
		has_research_agreement = event_target:ahw_sealer_$N$
		has_migration_access = event_target:ahw_sealer_$N$
		event_target:ahw_sealer_$N$ = { has_migration_access = prev }
		is_guaranteeing = event_target:ahw_sealer_$N$
		event_target:ahw_sealer_$N$ = { is_guaranteeing = prev }
		# at peace with each other, borders open both ways (user, 2026-11-05). Communications first:
		# unmet empires are no friends, whatever their border setting reads.
		AND = {
			has_communications = event_target:ahw_sealer_$N$
			NOT = { is_at_war_with = event_target:ahw_sealer_$N$ }
			NOT = { has_closed_borders = event_target:ahw_sealer_$N$ }
			event_target:ahw_sealer_$N$ = { NOT = { has_closed_borders = prev } }
		}
		# the same side of any war
		any_war = {
			OR = {
				AND = {
					is_war_participant = { who = prev side = attackers }
					is_war_participant = { who = event_target:ahw_sealer_$N$ side = attackers }
				}
				AND = {
					is_war_participant = { who = prev side = defenders }
					is_war_participant = { who = event_target:ahw_sealer_$N$ side = defenders }
				}
			}
		}
	}
}
