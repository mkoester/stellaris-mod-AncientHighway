# anchor's owner (user, 2026-11-04) studies the gate with a science ship, researches
# both gate techs and repairs it on the gate itself (the upgrade ahw_gate_broken_N -> ahw_gate_N,
# 0.1.0) - once, for good. The study is offered by the monthly check ahw.20. The former AHW_REPAIR_PROJECT
# was the pop-up's scripted repair; the contract's repair now ends in ahw_repair_gate directly.

special_project = {
	key = AHW_STUDY_PROJECT
	cost = 0
	days_to_research = 30
	tech_department = physics_technology
	picture = GFX_evt_engineering_research
	icon = "gfx/interface/icons/situation_log/situation_log_quest.dds"

	event_scope = ship_event
	requirements = { shipclass_science_ship = 1 }

	on_success = {
		# 2026-11-19: an AI nomad completed this (autosave 2226.08.01) but ahw.22 left no trace -
		# this line shows whether on_success runs and in which scope
		log = "AHW study project done: this [This.GetName], owner [This.Owner.GetName], date [GetDate]"
		ship_event = { id = ahw.22 }
	}
}

# The repair contract's work (ahw_contract_repair_gate), done by the nomad's arkship as vanilla's
# contract projects are. It only counts: the repair happens in the contract's on_success.
special_project = {
	key = "AHW_CONTRACT_REPAIR_PROJECT"
	event_chain = ahw_contract_repair_gate_chain
	days_to_research = 90
	project_type = project_type_contract
	picture = GFX_evt_ship_in_orbit_3
	icon = "gfx/interface/icons/situation_log/situation_log_quest.dds"
	location = yes
	remove_when_completed = yes

	event_scope = ship_event

	requirements = {
		carries_colony = 1
	}

	on_success = {
		log = "AHW contract work done on [GetDate] by [This.Owner.GetName]"
		owner = {
			add_mission_counter = {
				mission = ahw_contract_repair_gate
				counter = ahw_contract_repair_gate_counter
				amount = 1
			}
		}
	}

	on_fail = {
		log = "AHW contract work FAILED on [GetDate] for [This.Owner.GetName]"
		owner = { end_event_chain = ahw_contract_repair_gate_chain }
	}
}
