# SPIKE (Phase 0, 0.0.18): the outer ring round the galaxy, and a portal for every regular empire.
# ahw_spike.10 runs by itself at game start (common/on_actions/ahw_spike_on_actions.txt) and
# fires the window ahw_spike.1 for the human player.
# Console, with nothing selected (Esc) - see AGENTS.md, "Status":
#   event ahw_spike.2   report the state to game.log
#   event ahw_spike.3   allow us the Dimensional Lock order (as the astral action does)
#   event ahw_spike.4   lock our portal's gate for another country (direction test)
#   event ahw_spike.5   lock the ring end of our pair for another country
#   event ahw_spike.6   an outpost on our anchor (so we own it)
#   event ahw_spike.7   run the monthly study/repair check for us now
#   event ahw_spike.8   grant ahw_tech_spike
# Map changes reach route-finding on the next daily tick: let a day pass before ordering routes.
namespace = ahw_spike

# Builds everything. No scope (from on_game_start).
event = {
	id = ahw_spike.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = ahw_spike_done }
	}

	immediate = {
		set_global_flag = ahw_spike_done
		log = "AHW SPIKE ===== ahw_spike.10: building the highway at game start ====="

		ahw_spike_core_band = { FRACTION = 0.1 }
		ahw_spike_core_band = { FRACTION = 0.15 }
		ahw_spike_core_band = { FRACTION = 0.2 }

		# Run 8: spawn_system without a scope spawns nothing. Find the centre instead: two probe
		# systems leapfrog corewards from the innermost system (generated: ahw_spike_find_centre).
		# 0.0.14 (user, 2026-11-04): the ring goes round the outside of the galaxy, following its
		# outline - measured from the centre, then built (generated).
		ahw_spike_find_centre = yes
		log = "AHW SPIKE timing: centre found"
		ahw_spike_measure_outline = yes
		log = "AHW SPIKE timing: outline measured"
		ahw_spike_build_outer_ring = yes
		log = "AHW SPIKE timing: outer ring built"

		# A portal for every regular empire: a new system with one hyperlane, hanging off a system
		# 2-3 jumps from the capital (a random walk), holding the empire's entry gate; its exit is
		# in the nearest ring system. All repaired, for the spike.
		every_country = {
			limit = { is_country_type = default }
			clear_global_event_target = ahw_x_tmp
			clear_global_event_target = ahw_portal_tmp
			clear_global_event_target = ahw_node_tmp
			capital_star = { solar_system = { save_global_event_target_as = ahw_cap_tmp } }
			# Run 11 (user): search every system 2-3 jumps from the capital and take one from the best
			# tier: fewest lanes first, systems that already hold a bypass after those without.
			if = {
				limit = { any_system = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 1 } NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes } }
				random_system = {
					limit = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 1 } NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes }
					save_global_event_target_as = ahw_x_tmp
				}
				log = "AHW SPIKE anchor for [This.GetName] (capital [ahw_cap_tmp.GetName]): tier 1 (a dead end)"
			}
			else_if = {
				limit = { any_system = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 2 } NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes } }
				random_system = {
					limit = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 2 } NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes }
					save_global_event_target_as = ahw_x_tmp
				}
				log = "AHW SPIKE anchor for [This.GetName] (capital [ahw_cap_tmp.GetName]): tier 2 (at most two lanes)"
			}
			else_if = {
				limit = { any_system = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 2 } ahw_spike_within_3_jumps_of_capital = yes } }
				random_system = {
					limit = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 2 } ahw_spike_within_3_jumps_of_capital = yes }
					save_global_event_target_as = ahw_x_tmp
				}
				log = "AHW SPIKE anchor for [This.GetName] (capital [ahw_cap_tmp.GetName]): tier 3 (at most two lanes, but a bypass)"
			}
			else_if = {
				limit = { any_system = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 3 } NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes } }
				random_system = {
					limit = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 3 } NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes }
					save_global_event_target_as = ahw_x_tmp
				}
				log = "AHW SPIKE anchor for [This.GetName] (capital [ahw_cap_tmp.GetName]): tier 4 (at most three lanes)"
			}
			else_if = {
				limit = { any_system = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 3 } ahw_spike_within_3_jumps_of_capital = yes } }
				random_system = {
					limit = { ahw_spike_portal_anchor_ok = yes count_neighbor_system = { count <= 3 } ahw_spike_within_3_jumps_of_capital = yes }
					save_global_event_target_as = ahw_x_tmp
				}
				log = "AHW SPIKE anchor for [This.GetName] (capital [ahw_cap_tmp.GetName]): tier 5 (at most three lanes, but a bypass)"
			}
			else_if = {
				limit = { any_system = { ahw_spike_portal_anchor_ok = yes NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes } }
				random_system = {
					limit = { ahw_spike_portal_anchor_ok = yes NOT = { ahw_spike_has_bypass = yes } ahw_spike_within_3_jumps_of_capital = yes }
					save_global_event_target_as = ahw_x_tmp
				}
				log = "AHW SPIKE anchor for [This.GetName] (capital [ahw_cap_tmp.GetName]): tier 6 (more lanes)"
			}
			else_if = {
				limit = { any_system = { ahw_spike_portal_anchor_ok = yes ahw_spike_within_3_jumps_of_capital = yes } }
				random_system = {
					limit = { ahw_spike_portal_anchor_ok = yes ahw_spike_within_3_jumps_of_capital = yes }
					save_global_event_target_as = ahw_x_tmp
				}
				log = "AHW SPIKE anchor for [This.GetName] (capital [ahw_cap_tmp.GetName]): tier 7 (more lanes and a bypass)"
			}
			if = {
				limit = { exists = event_target:ahw_x_tmp }
				# Run 8 (user): the portal's lane must arrive in the anchor system away from its
				# other lanes, so an attacker from the portal has to pass the anchor's starbase.
				# 0.0.17: spawned there at once, the angle computed from coordinates (generated).
				ahw_spike_place_portal = yes
				add_hyperlane = { from = event_target:ahw_portal_tmp to = event_target:ahw_x_tmp }
				ahw_spike_find_nearest_node = yes
				ahw_spike_make_next_pair = yes
				# 0.0.18: broken until studied and repaired by the anchor's owner (ahw_spike.20-.23);
				# the study teaches the types and the ring (runs 4-5).
				event_target:ahw_portal_tmp = { set_star_flag = ahw_broken }
				set_visited = event_target:ahw_portal_tmp
				if = {
					limit = { is_ai = no }
					event_target:ahw_portal_tmp = { save_global_event_target_as = ahw_player_portal }
					event_target:ahw_node_tmp = { save_global_event_target_as = ahw_player_node }
					event_target:ahw_x_tmp = { save_global_event_target_as = ahw_player_anchor }
				}
				log = "AHW SPIKE portal for [This.GetName] (capital [ahw_cap_tmp.GetName]): off [ahw_x_tmp.GetName], exit in [ahw_node_tmp.GetName]"
			}
			else = {
				log = "AHW SPIKE NO portal for [This.GetName]: the random walk found no system 2-3 jumps out"
			}
		}

		every_country = {
			limit = { is_ai = no }
			ahw_spike_report = { WHEN = at_game_start }
			country_event = { id = ahw_spike.1 }
		}
	}
}

country_event = {
	id = ahw_spike.1
	title = ahw_spike.1.name
	desc = ahw_spike.1.desc
	is_triggered_only = yes
	option = { name = ahw_spike.ok }
}

# Direction test: lock our portal's gate for ANOTHER country (360 days), so our own ships are the
# outsiders. Can they still leave the portal through the gate, and come back from the ring?
country_event = {
	id = ahw_spike.4
	title = ahw_spike.4.name
	desc = ahw_spike.4.desc
	is_triggered_only = yes
	immediate = {
		random_country = {
			limit = { is_country_type = default NOT = { is_same_value = root } }
			save_event_target_as = ahw_other
		}
		event_target:ahw_player_portal = {
			every_bypass_in_system = {
				lock_bypass = { country = event_target:ahw_other duration = 360 }
			}
		}
		log = "AHW SPIKE our portal's gate locked for [ahw_other.GetName]"
		ahw_spike_report = { WHEN = after_foreign_lock }
	}
	option = { name = ahw_spike.ok }
}

# An outpost on our anchor, so we own it (the anchor's owner studies, repairs and locks).
country_event = {
	id = ahw_spike.6
	title = ahw_spike.6.name
	desc = ahw_spike.6.desc
	is_triggered_only = yes
	immediate = {
		event_target:ahw_player_anchor = {
			if = {
				limit = { NOT = { exists = owner } }
				create_starbase = { size = starbase_outpost owner = root }
			}
			log = "AHW SPIKE anchor [This.GetName]: outpost created for [Root.GetName]"
		}
	}
	option = { name = ahw_spike.ok }
}

# Monthly (on_monthly_pulse_country): for each portal whose anchor this country owns, offer the
# study (science ship special project), then - studied and with the tech - the repair (ahw_spike.21).
country_event = {
	id = ahw_spike.20
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_country_type = default }
	immediate = {
		every_system = {
			limit = {
				has_star_flag = ahw_portal
				any_neighbor_system = { exists = owner owner = { is_same_value = root } }
			}
			if = {
				limit = {
					NOT = { has_star_flag = ahw_studied }
					NOT = { has_star_flag = ahw_study_offered }
				}
				set_star_flag = ahw_study_offered
				enable_special_project = { name = AHW_SPIKE_STUDY_PROJECT location = this owner = root }
				log = "AHW SPIKE study of the entry point in [This.GetName] offered to [Root.GetName]"
			}
			else_if = {
				limit = {
					has_star_flag = ahw_studied
					has_star_flag = ahw_broken
					NOT = { has_star_flag = ahw_repair_offered }
					root = { has_technology = ahw_tech_spike }
				}
				set_star_flag = ahw_repair_offered
				save_event_target_as = ahw_repair_portal
				root = { country_event = { id = ahw_spike.21 } }
				log = "AHW SPIKE repair of the entry point in [This.GetName] offered to [Root.GetName]"
			}
		}
	}
}

# The repair offer: pay alloys, then a construction ship repairs the gate. "Not now" offers it
# again in a year. Special projects cannot charge resources themselves.
country_event = {
	id = ahw_spike.21
	title = ahw_spike.21.name
	desc = ahw_spike.21.desc
	is_triggered_only = yes
	option = {
		name = ahw_spike.21.a
		allow = { has_resource = { type = alloys amount >= 500 } }
		add_resource = { alloys = -500 }
		event_target:ahw_repair_portal = {
			enable_special_project = { name = AHW_SPIKE_REPAIR_PROJECT location = this owner = root }
		}
		ai_chance = { factor = 10 }
	}
	option = {
		name = ahw_spike.21.b
		event_target:ahw_repair_portal = {
			remove_star_flag = ahw_repair_offered
			set_timed_star_flag = { flag = ahw_repair_offered days = 360 }
		}
		ai_chance = { factor = 1 }
	}
}

# Study done (this = the science ship, in the portal): the country knows the network and the ring.
ship_event = {
	id = ahw_spike.22
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		solar_system = { set_star_flag = ahw_studied }
		owner = {
			set_country_flag = ahw_highway_studied
			ahw_spike_learn_types = yes
			every_system = {
				limit = { has_star_flag = ahw_ring }
				prev = { set_visited = prev }
			}
		}
		log = "AHW SPIKE entry point in [This.System.GetName] studied by [This.Owner.GetName]"
	}
}

	immediate = {
		solar_system = { remove_star_flag = ahw_broken }
