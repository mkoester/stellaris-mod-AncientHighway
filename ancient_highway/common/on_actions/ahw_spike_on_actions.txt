# SPIKE (Phase 0, 0.0.8): build the highway at game start, without a scope (as vanilla's
# game_start.31 creates the wormhole pairs from on_game_start). ahw_spike.10 then fires
# ahw_spike.1 (the window) for the human player.
on_game_start = {
	events = {
		ahw_spike.10
	}
}

# 0.0.31: the mod added to a running game builds the highway on the save's load (single player)
# or the next month (multiplayer, where the load hook does not fire); ahw_spike.10's trigger lets
# it run once.
on_single_player_save_game_load = {
	events = {
		ahw_spike.10
	}
}
on_monthly_pulse = {
	events = {
		ahw_spike.10

	}
