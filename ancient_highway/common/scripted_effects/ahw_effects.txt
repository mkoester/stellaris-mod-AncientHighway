# Everything logs to game.log with the prefix "AHW".
# In effects with parameters, "[" in a log string must be written "\\[" (run 1).
# A $PARAM$ next to "\\[" comes out mangled (run 2), so parameters go last in each log line.

# One ring node (event target $NAME$, named $LABEL$), $R$ from the scope (no scope: the galaxy's origin?), direction $ANGLE$.
ahw_spawn_ring_node = {
	spawn_system = {
		min_distance = $R$
		max_distance = $R$
		max_jumps = 0
		min_orientation_angle = $ANGLE$
		max_orientation_angle = $ANGLE$
		initializer = ahw_ring_init
		hyperlane = no
		is_discovered = yes
		effect = {
			set_star_flag = ahw_ring
			set_star_flag = ahw_unclaimable
			# inlined, not ahw_rename_system (run 22: one more level of scripted effects passed
			# the engine's limit of 5, "Max effects post init recursive depth")
			set_name = $LABEL$
			every_system_planet = { limit = { is_star = yes } set_name = $LABEL$ }
			save_global_event_target_as = $NAME$
		}
	}
}

# this = system: its distance_to_core_percent into ahw_d.
ahw_measure = {
	export_trigger_value_to_variable = {
		trigger = distance_to_core_percent
		variable = ahw_d
	}
}

# One leapfrog step towards the centre: probe T moves $S$ corewards of probe A; if T is now closer
# to the centre, the two swap roles (ahw_probe_a is always the best so far).
ahw_probe_step = {
	event_target:ahw_probe_t = {
		move_system = {
			target = event_target:ahw_probe_a
			min_distance >= $S$
			max_distance <= $S$
			max_jumps = 0
			direction = corewards
			hyperlane = no
		}
		ahw_measure = yes
	}
	if = {
		limit = {
			event_target:ahw_probe_t = { check_variable = { which = ahw_d value < event_target:ahw_probe_a.ahw_d } }
		}
		event_target:ahw_probe_t = { save_global_event_target_as = ahw_probe_swap }
		event_target:ahw_probe_a = { save_global_event_target_as = ahw_probe_t }
		event_target:ahw_probe_swap = { save_global_event_target_as = ahw_probe_a }
	}
}

# this = a portal: locks both ends of its pair (the gate here, the exit in the ring) for $COUNTRY$,
# this = a system: renamed to $KEY$, with its star (run 21: the system view's label is the star's
# name, so renaming only the system left "Highway Portal" there, also after a reload; vanilla
# renames both, events/leviathans_events_2.txt).
ahw_rename_system = {
	set_name = $KEY$
	every_system_planet = {
		limit = { is_star = yes }
		set_name = $KEY$
	}
}

# this = the gate megastructure (overclock mode Sealed): both ends closed to all but its owner.
# 0.0.23: by flags the pair's bypass type checks (country_can_use), not lock_bypass - run 21: a
# 1-day lock_bypass never expired, so Open could not undo a seal.
ahw_seal_gate = {
	if = {
		limit = { exists = owner }
		owner = { save_event_target_as = ahw_sealer }
		solar_system = {
			set_star_flag = ahw_sealed
			ahw_seal_pair_for = { COUNTRY = event_target:ahw_sealer }
			log = "AHW gate in \\[This.GetName] SEALED for \\[ahw_sealer.GetName]"
		}
	}
}

# this = the gate megastructure (overclock mode Open): if it was sealed, open to all again. Also
# runs when the mode is first set at creation, hence the flag check.
ahw_unseal_gate = {
	if = {
		limit = {
			exists = owner
			solar_system = { has_star_flag = ahw_sealed }
		}
		owner = { save_event_target_as = ahw_sealer }
		solar_system = {
			remove_star_flag = ahw_sealed
			ahw_unseal_pair = yes
			log = "AHW gate in \\[This.GetName] opened by \\[ahw_sealer.GetName]"
		}
	}
}

# What the ring nodes and portals are now (the result, not the dice): counts on ring node 1, logged.
ahw_count_centres = {
	event_target:ahw_ring_1 = {
		set_variable = { which = ahw_n_bh value = 0 }
		set_variable = { which = ahw_n_pulsar value = 0 }
		set_variable = { which = ahw_n_toxoid value = 0 }
		set_variable = { which = ahw_n_neutron value = 0 }
		set_variable = { which = ahw_n_other value = 0 }
		set_variable = { which = ahw_n_debris value = 0 }
		set_variable = { which = ahw_n_deposit value = 0 }
	}
	every_system = {
		limit = { OR = { has_star_flag = ahw_ring has_star_flag = ahw_portal } }
		if = {
			limit = { is_star_class = ahw_sc_black_hole }
			event_target:ahw_ring_1 = { change_variable = { which = ahw_n_bh value = 1 } }
		}
		else_if = {
			limit = { is_star_class = ahw_sc_pulsar }
			event_target:ahw_ring_1 = { change_variable = { which = ahw_n_pulsar value = 1 } }
		}
		else_if = {
			limit = { is_star_class = ahw_sc_toxoid_star }
			event_target:ahw_ring_1 = { change_variable = { which = ahw_n_toxoid value = 1 } }
		}
		else_if = {
			limit = { is_star_class = ahw_sc_neutron_star has_star_flag = ahw_portal }
			event_target:ahw_ring_1 = { change_variable = { which = ahw_n_neutron value = 1 } }
		}
		else = {
			event_target:ahw_ring_1 = { change_variable = { which = ahw_n_other value = 1 } }
		}
		every_system_planet = {
			limit = { has_planet_flag = ahw_debris }
			event_target:ahw_ring_1 = { change_variable = { which = ahw_n_debris value = 1 } }
		}
		every_system_planet = {
			limit = { has_planet_flag = ahw_debris_deposit }
			event_target:ahw_ring_1 = { change_variable = { which = ahw_n_deposit value = 1 } }
		}
	}
	event_target:ahw_ring_1 = {
		log = "AHW centres: black hole \\[This.ahw_n_bh], pulsar \\[This.ahw_n_pulsar], toxoid \\[This.ahw_n_toxoid], neutron (portals only) \\[This.ahw_n_neutron], other \\[This.ahw_n_other] (ring nodes and portals)"
		log = "AHW debris: \\[This.ahw_n_debris] asteroids, \\[This.ahw_n_deposit] with a deposit (ring nodes and portals)"
	}
}

# 0.0.31: one portal for the country in scope (this = country; "[" in its logs escaped too - run 31:
# "Invalid macro entry in ahw_build_portal", although it takes no parameters), from the game-start build and,
# for empires born later or a build in a running game, from ahw.30. The caller holds the
# spawn batch. The anchor: an unowned system first; else (a running game, where most systems near
# a capital are owned) one the empire owns itself, never another's (user, 2026-11-06).
ahw_build_portal = {
	clear_global_event_target = ahw_x_tmp
	clear_global_event_target = ahw_portal_tmp
	clear_global_event_target = ahw_node_tmp
	capital_star = { solar_system = { save_global_event_target_as = ahw_cap_tmp } }
	save_global_event_target_as = ahw_country_tmp
	ahw_pick_anchor = { OK = ahw_portal_anchor_ok WHO = unowned }
	if = {
		limit = { NOT = { exists = event_target:ahw_x_tmp } }
		ahw_pick_anchor = { OK = ahw_portal_anchor_own_ok WHO = own }
	}
	if = {
		limit = { exists = event_target:ahw_x_tmp }
		# Run 8 (user): the portal's lane must arrive in the anchor system away from its
		# other lanes, so an attacker from the portal has to pass the anchor's starbase.
		# 0.0.17: spawned there at once, the angle computed from coordinates (generated).
		ahw_place_portal = yes
		add_hyperlane = { from = event_target:ahw_portal_tmp to = event_target:ahw_x_tmp }
		ahw_find_nearest_node = yes
		ahw_make_next_pair = yes
		set_country_flag = ahw_has_portal
		# 0.0.21: named after the ring's reach it leads into, with an octal address.
		ahw_name_portal = yes
		# 0.0.18: broken until studied and repaired by the anchor's owner (ahw.20-.23);
		# the study teaches the types and the ring (runs 4-5).
		event_target:ahw_portal_tmp = {
			set_star_flag = ahw_broken
			# 0.1.0 (run 43): which empire the portal was built for (console ahw_debug.74)
			set_star_flag = ahw_portal_for@event_target:ahw_country_tmp
		}
		set_visited = event_target:ahw_portal_tmp
		if = {
			limit = { is_ai = no }
			event_target:ahw_portal_tmp = { save_global_event_target_as = ahw_player_portal }
			event_target:ahw_node_tmp = { save_global_event_target_as = ahw_player_node }
			event_target:ahw_x_tmp = { save_global_event_target_as = ahw_player_anchor }
		}
		log = "AHW portal for \\[This.GetName] (capital \\[ahw_cap_tmp.GetName]): off \\[ahw_x_tmp.GetName], exit in \\[ahw_node_tmp.GetName]"
	}
	else = {
		log = "AHW NO portal for \\[This.GetName]: the random walk found no unowned or own system 2-3 jumps out"
	}
}

		}

	}
}

# this = a portal, event_target:ahw_studier = the country: the study's result (ahw.22, and
# the console skip ahw_debug.40 - one code path, so the skip tests the real thing).
ahw_mark_studied = {
	# the portal's flag stands for its anchor owner's study (ahw.20 offers it until set); a
	# nomad's study (ahw.60) teaches the nomad and leaves the owner's offer alone
	if = {
		limit = { event_target:ahw_studier = { is_nomadic = no } }
		set_star_flag = ahw_studied
		# run 40: the gate is the anchor owner's at once, not from the next monthly check (the
		# console skip studied before it, and the upgrade showed a disabled "Complete" until then)
		if = {
			limit = { any_neighbor_system = { exists = owner owner = { is_same_value = event_target:ahw_studier } } }
			every_system_megastructure = { set_owner = event_target:ahw_studier }
		}
	}
	ahw_country_learns = yes
	# 0.0.38 (design 2026-11-05): the study grants 15 % of each tech, once per empire.
	event_target:ahw_studier = { ahw_grant = { FLAG = ahw_grant_study PROGRESS = 0.15 } }
	log = "AHW entry point in \\[This.GetName] studied by \\[ahw_studier.GetName]"
}

# this = a portal, event_target:ahw_studier = the country: what the country knows after studying
# this gate, or after the head start at it (ahw.20) - no grant.
ahw_country_learns = {
	event_target:ahw_studier = {
		set_country_flag = ahw_highway_studied
		# Run 18: the tech waited for the next physics draw. Vanilla does this after studying a
		# ruined gateway (tech_gateway_activation): the option appears at once and stays.
		add_research_option = ahw_tech_gate_theory
		add_research_option = ahw_tech_gate_construction
		ahw_learn_types = yes
	}
	# 0.0.39: the own portal's centre counts as seen (physics grants for other kinds only).
	ahw_see_kind = { WHO = event_target:ahw_studier }
	# 0.0.26 (design 2026-11-06): only this gate's exit node, not the whole ring - the rest is
	# explored by flying it.
	ahw_reveal_exit = yes
}

# this = a country. A share of both techs, once per empire (FLAG) - the study and the head start
# (design 2026-11-05).
ahw_grant = {
	if = {
		limit = { NOT = { has_country_flag = $FLAG$ } }
		set_country_flag = $FLAG$
		ahw_grant_tech = { TECH = ahw_tech_gate_theory VAR = ahw_got_theory PROGRESS = $PROGRESS$ }
		ahw_grant_tech = { TECH = ahw_tech_gate_construction VAR = ahw_got_construction PROGRESS = $PROGRESS$ }
		log = "AHW grant $FLAG$ ($PROGRESS$ of each tech) to \\[This.GetName]"
	}
}

# this = a country. PROGRESS of TECH (a fraction of its cost, add_tech_progress), but all grants
# of a tech together at most 0.8 (0.0.39, design 2026-11-06), counted in the country variable VAR;
# nothing for a tech already researched.
ahw_grant_tech = {
	if = {
		limit = { NOT = { has_technology = $TECH$ } }
		# run 40: unset on the first grant ("Variable ... is not set" in error.log)
		if = {
			limit = { NOT = { is_variable_set = $VAR$ } }
			set_variable = { which = $VAR$ value = 0 }
		}
		set_variable = { which = ahw_g value = 0.8 }
		subtract_variable = { which = ahw_g value = $VAR$ }
		if = {
			limit = { check_variable = { which = ahw_g value > $PROGRESS$ } }
			set_variable = { which = ahw_g value = $PROGRESS$ }
		}
		if = {
			limit = { check_variable = { which = ahw_g value > 0 } }
			add_tech_progress = { tech = $TECH$ progress = ahw_g }
			change_variable = { which = $VAR$ value = ahw_g }
		}
		ahw_cap_check = yes
		# (a $PARAM$ inside a string is mangled, run 36: the tech's name stays readable, a [This.$VAR$] would not)
		log = "AHW \\[This.GetName]: +\\[This.ahw_g] of $TECH$ (asked $PROGRESS$); granted so far: theory \\[This.ahw_got_theory], construction \\[This.ahw_got_construction]"
	}
}

# this = a country (after a grant, and on_tech_increased): once a tech is capped (0.8 granted) or
# researched, its open inspection studies are withdrawn and no new ones offered (0.0.40, user
# 2026-11-07).
ahw_cap_check = {
	if = {
		limit = {
			NOT = { has_country_flag = ahw_closed_construction }
			OR = {
				has_technology = ahw_tech_gate_construction
				check_variable = { which = ahw_got_construction value >= 0.8 }
			}
		}
		set_country_flag = ahw_closed_construction
		country_event = { id = ahw.66 }	# withdraws the studies (an event: see tools/gen.py gen_events)
		log = "AHW \\[This.GetName]: Gate Construction capped or researched - gate studies withdrawn"
	}
	if = {
		limit = {
			NOT = { has_country_flag = ahw_closed_theory }
			OR = {
				has_technology = ahw_tech_gate_theory
				check_variable = { which = ahw_got_theory value >= 0.8 }
			}
		}
		set_country_flag = ahw_closed_theory
		country_event = { id = ahw.67 }
		log = "AHW \\[This.GetName]: Gate Theory capped or researched - centre studies withdrawn"
	}
}

# this = a portal, event_target:ahw_insp_country = the country (ahw_inspect_centre, after a
# centre study): Gate
# Theory +7 % for a centre of a kind the country has not seen, else +5 % (design 2026-11-06).
ahw_centre_grant = {
	if = {
		limit = { ahw_kind_seen = { WHO = event_target:ahw_insp_country } }
		event_target:ahw_insp_country = {
			ahw_grant_tech = { TECH = ahw_tech_gate_theory VAR = ahw_got_theory PROGRESS = 0.05 }
		}
	}
	else = {
		event_target:ahw_insp_country = {
			ahw_grant_tech = { TECH = ahw_tech_gate_theory VAR = ahw_got_theory PROGRESS = 0.07 }
		}
	}
	ahw_see_kind = { WHO = event_target:ahw_insp_country }
	log = "AHW centre of \\[This.GetName] inspected by \\[ahw_insp_country.GetName]"
}

# this = a portal: WHO has now seen its centre's kind (country flags ahw_kind_<class>).
ahw_see_kind = {
	if = {
		limit = { is_star_class = ahw_sc_black_hole }
		$WHO$ = { set_country_flag = ahw_kind_black_hole }
	}
	else_if = {
		limit = { is_star_class = ahw_sc_pulsar }
		$WHO$ = { set_country_flag = ahw_kind_pulsar }
	}
	else_if = {
		limit = { is_star_class = ahw_sc_toxoid_star }
		$WHO$ = { set_country_flag = ahw_kind_toxoid }
	}
	else_if = {
		limit = { is_star_class = ahw_sc_neutron_star }
		$WHO$ = { set_country_flag = ahw_kind_neutron }
	}
}

# this = a portal, event_target:ahw_repairer = the country: the repair's result - from the
# repaired gate's on_build_complete, or ahw_repair_gate (script). Runs once: a second call (the
# scripted path's fallback) finds the portal no longer broken.
ahw_mark_repaired = {
	if = {
		limit = { has_star_flag = ahw_broken }
		remove_star_flag = ahw_broken
		# The gate belongs to the anchor's owner (its Open/Sealed switch); with the anchor
		# unowned, to a settled repairer - a nomad never holds it (design: nomads never seal).
		if = {
			limit = { any_neighbor_system = { exists = owner } }
			random_neighbor_system = {
				limit = { exists = owner }
				owner = { save_event_target_as = ahw_gate_owner }
			}
			every_system_megastructure = { set_owner = event_target:ahw_gate_owner }
		}
		else_if = {
			limit = { event_target:ahw_repairer = { is_nomadic = no } }
			every_system_megastructure = { set_owner = event_target:ahw_repairer }
		}
		# 0.0.38: the head start is for a gate somebody else repaired, never for one's own.
		event_target:ahw_repairer = { set_country_flag = ahw_grant_head_start }
		# 0.0.39: nor the inspection grant for it.
		ahw_flag_repairer = yes
		log = "AHW entry point in \\[This.GetName] repaired by \\[ahw_repairer.GetName]"
	}
}

# 0.1.1, shared charts (design "Exploration and incentives"): every ring node charted by
# event_target:ahw_chart_from (star flag ahw_mapped@, set on entry in ahw.60) becomes known to
# event_target:ahw_chart_to - visited (set_visited, as the study reveals the exit node) and charted.
ahw_copy_charts = {
	event_target:ahw_chart_to = { set_variable = { which = ahw_n_copied value = 0 } }
	every_system = {
		limit = {
			has_star_flag = ahw_ring
			has_star_flag = ahw_mapped@event_target:ahw_chart_from
			NOT = { has_star_flag = ahw_mapped@event_target:ahw_chart_to }
		}
		set_star_flag = ahw_mapped@event_target:ahw_chart_to
		event_target:ahw_chart_to = {
			set_visited = prev
			change_variable = { which = ahw_n_copied value = 1 }
		}
	}
	if = {
		limit = { event_target:ahw_chart_to = { check_variable = { which = ahw_n_copied value > 0 } } }
		event_target:ahw_chart_to = {
			log = "AHW charts: \\[This.ahw_n_copied] ring nodes from \\[ahw_chart_from.GetName] to \\[This.GetName]"
		}
	}
}

# No scope (yearly, ahw.80): close friends (ahw_close_friend) give each other their charts. A
# friend's friend may get them in the same pass, via the friend - as it would a year later.
ahw_share_charts_friends = {
	every_country = {
		limit = { is_country_type = default }
		save_event_target_as = ahw_chart_from
		every_country = {
			limit = {
				is_country_type = default
				NOT = { is_same_value = event_target:ahw_chart_from }
				ahw_close_friend = { WHO = event_target:ahw_chart_from }
			}
			save_event_target_as = ahw_chart_to
			ahw_copy_charts = yes
		}
	}
}

# No scope: every Galactic Community member gets every member's charts (resolution Share Ring
# Charts: on passing, then yearly in ahw.80). Pools them on a temporary star flag first.
ahw_share_charts_gc = {
	every_country = {
		limit = { is_galactic_community_member = yes }
		save_event_target_as = ahw_chart_from
		every_system = {
			limit = { has_star_flag = ahw_ring has_star_flag = ahw_mapped@event_target:ahw_chart_from }
			set_star_flag = ahw_gc_pool
		}
	}
	every_country = {
		limit = { is_galactic_community_member = yes }
		save_event_target_as = ahw_chart_to
		set_variable = { which = ahw_n_copied value = 0 }
		every_system = {
			limit = {
				has_star_flag = ahw_gc_pool
				NOT = { has_star_flag = ahw_mapped@event_target:ahw_chart_to }
			}
			set_star_flag = ahw_mapped@event_target:ahw_chart_to
			event_target:ahw_chart_to = {
				set_visited = prev
				change_variable = { which = ahw_n_copied value = 1 }
			}
		}
		if = {
			limit = { check_variable = { which = ahw_n_copied value > 0 } }
			log = "AHW charts: \\[This.ahw_n_copied] ring nodes from the Galactic Community to \\[This.GetName]"
		}
	}
	every_system = {
		limit = { has_star_flag = ahw_gc_pool }
		remove_star_flag = ahw_gc_pool
	}
}
