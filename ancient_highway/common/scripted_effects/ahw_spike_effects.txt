# SPIKE (Phase 0). Everything logs to game.log with the prefix "AHW SPIKE".
# In effects with parameters, "[" in a log string must be written "\\[" (run 1).
# A $PARAM$ next to "\\[" comes out mangled (run 2), so parameters go last in each log line.

# this = system
ahw_spike_log_system = {
	export_trigger_value_to_variable = {
		trigger = distance_to_core_percent
		variable = ahw_core_pct
	}
	log = "AHW SPIKE \\[This.GetName]: distance_to_core_percent (a fraction) = \\[This.ahw_core_pct] -- $WHAT$"
	if = {
		limit = { has_star_flag = ahw_ring }
		log = "AHW SPIKE   \\[This.GetName]: ring system"
	}
	if = {
		limit = { has_star_flag = ahw_nexus_1 }
		log = "AHW SPIKE   \\[This.GetName]: nexus of ahw_spike_entry_1"
	}
	if = {
		limit = { has_star_flag = ahw_entry_open_1 }
		log = "AHW SPIKE   \\[This.GetName]: entry point OPEN (node of ahw_spike_entry_1)"
	}
	if = {
		limit = { exists = owner }
		log = "AHW SPIKE   \\[This.GetName]: owned by \\[This.Owner.GetName]"
	}
	every_bypass_in_system = {
		if = {
			limit = { is_bypass_type = ahw_spike_entry_1 }
			log = "AHW SPIKE   bypass ahw_spike_entry_1"
		}
		else_if = {
			limit = { is_bypass_type = ahw_spike_shortcut }
			log = "AHW SPIKE   bypass ahw_spike_shortcut"
		}
		else = {
			log = "AHW SPIKE   bypass of another type"
		}
		if = {
			limit = { has_lock = yes }
			log = "AHW SPIKE     ...LOCKED"
		}
	}
	if = {
		limit = {
			any_system_megastructure = {
				OR = {
					is_megastructure_type = ahw_spike_entry_gate_1
					is_megastructure_type = ahw_spike_entry_ring_1
				}
			}
		}
		log = "AHW SPIKE   entry-point megastructure present"
	}
}

# Spawns ring system $NAME$ $DIST$ away from $FROM$, in direction $ANGLE$, no hyperlane.
# Run 2: the angle is an absolute map direction.
ahw_spike_spawn_ring_step = {
	event_target:$FROM$ = {
		spawn_system = {
			min_distance = $DIST$
			max_distance = $DIST$
			max_jumps = 0
			min_orientation_angle = $ANGLE$
			max_orientation_angle = $ANGLE$
			initializer = ahw_spike_ring_init
			hyperlane = no
			is_discovered = yes
			effect = {
				set_star_flag = ahw_ring
				set_name = $NAME$ # loc key: "Highway Node N" (run 5: all six looked alike)
				save_global_event_target_as = $NAME$
			}
		}
	}
}

# How empty is the core? distance_to_core_percent is a fraction 0-1 (run 2).
ahw_spike_core_band = {
	if = {
		limit = { any_system = { distance_to_core_percent < $FRACTION$ } }
		log = "AHW SPIKE   some systems are closer to the centre than $FRACTION$"
	}
	else = {
		log = "AHW SPIKE   NO system is closer to the centre than $FRACTION$"
	}
}

# this = country (the player)
ahw_spike_report = {
	log = "AHW SPIKE ===== report: $WHEN$ ====="
	if = {
		limit = { has_technology = ahw_tech_spike }
		log = "AHW SPIKE \\[This.GetName] has ahw_tech_spike"
	}
	else = {
		log = "AHW SPIKE \\[This.GetName] does NOT have ahw_tech_spike"
	}
	event_target:ahw_anchor = { ahw_spike_log_system = { WHAT = anchor_A_entry_open } }
	event_target:ahw_anchor_b = { ahw_spike_log_system = { WHAT = system_B_entry_open } }
	event_target:ahw_anchor_c = { ahw_spike_log_system = { WHAT = system_C_entry_broken } }
	event_target:ahw_ring_1 = { ahw_spike_log_system = { WHAT = ring_1_nexus } }
	event_target:ahw_ring_4 = { ahw_spike_log_system = { WHAT = ring_4 } }
	every_system = {
		limit = { has_star_flag = ahw_ring }
		ahw_spike_log_system = { WHAT = ring_any }
	}
}
