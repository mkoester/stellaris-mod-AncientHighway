# SPIKE (Phase 0, 0.0.24): the outer ring round the galaxy, and a portal for every regular empire.
# ahw_spike.10 runs by itself at game start (common/on_actions/ahw_spike_on_actions.txt) and
# fires the window ahw_spike.1 for the human player.
# Console, with nothing selected (Esc) - see AGENTS.md, "Status":
#   event ahw_spike.2   report the state to game.log
#   event ahw_spike.6   an outpost on our anchor (so we own it)
#   event ahw_spike.7   run the monthly study/repair check for us now
#   event ahw_spike.8   grant ahw_tech_spike
#   event ahw_spike.25  seal our pair against everyone, us included (again: open it)
#   event ahw_spike.26  seal our pair for another empire (a stranger to us)
#   event ahw_spike.27  make that empire our subject (now a friend of the sealer)
# Map changes reach route-finding on the next daily tick: let a day pass before ordering routes.
namespace = ahw_spike

# Builds everything, once. No scope: from on_game_start, or - the mod added to a running game
# (0.0.31) - from on_single_player_save_game_load or on_monthly_pulse.
event = {
	id = ahw_spike.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = ahw_spike_done }
	}

	immediate = {
		set_global_flag = ahw_spike_done
		log = "AHW SPIKE ===== ahw_spike.10: building the highway at game start ====="

		# 0.0.30 (run 27: a spawn costs ~N^2.7): vanilla batches its spawns so "caches are
		# recalculated only once rather than for every system spawned" (distar.290, the L-Cluster;
		# also for adding and removing hyperlanes). Everything up to the portals runs in one batch.
		set_spawn_system_batch = begin
		ahw_spike_core_band = { FRACTION = 0.1 }
		ahw_spike_core_band = { FRACTION = 0.15 }
		ahw_spike_core_band = { FRACTION = 0.2 }

		# Run 8: spawn_system without a scope spawns nothing. Find the centre instead: two probe
		# systems leapfrog corewards from the innermost system (generated: ahw_spike_find_centre).
		# 0.0.14 (user, 2026-11-04): the ring goes round the outside of the galaxy, following its
		# outline - measured from the centre, then built (generated).
		ahw_spike_find_centre = yes
		log = "AHW SPIKE timing: centre found"
		ahw_spike_measure_outline = yes
		log = "AHW SPIKE timing: outline measured"
		# 0.0.29: the galaxy's size in systems, to fit the cost per spawn against it.
		event_target:ahw_centre_system = {
			set_variable = { which = ahw_n_sys value = 0 }
			every_system = {
				limit = { NOT = { has_star_flag = ahw_unclaimable } }
				prev = { change_variable = { which = ahw_n_sys value = 1 } }
			}
			log = "AHW SPIKE systems: [This.ahw_n_sys] regular systems"
		}
		ahw_spike_build_outer_ring = yes
		log = "AHW SPIKE timing: outer ring built"

		# A portal for every regular empire (ahw_spike_build_portal; 0.0.31: the same effect serves
		# empires born later, ahw_spike.30).
		every_country = {
			limit = { is_country_type = default }
			ahw_spike_build_portal = yes
		}

		set_spawn_system_batch = end
		log = "AHW SPIKE timing: spawn batch ended"

		# 0.0.26: a third of the centres stay black holes, the rest become pulsars or toxoid stars;
		# 0.0.32: and debris in each
		ahw_spike_pick_centres = yes
		log = "AHW SPIKE timing: centres and debris done"

		every_country = {
			limit = { is_ai = no }
			ahw_spike_report = { WHEN = at_game_start }
			country_event = { id = ahw_spike.1 }
		}
	}
}

# Monthly (on_monthly_pulse_country): for each portal whose anchor this country owns, offer the
# study (science ship special project), then - studied and with the tech - the repair (ahw_spike.21).
country_event = {
	id = ahw_spike.20
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_country_type = default }
	immediate = {
		every_system = {
			limit = {
				has_star_flag = ahw_portal
				any_neighbor_system = { exists = owner owner = { is_same_value = root } }
			}
			# 0.0.20: the gate belongs to the anchor's owner (its Open/Sealed switch shows for the
			# owner only); a sealed gate stays locked for the current owner (renewed, so it also
			# follows a change of hands).
			every_system_megastructure = {
				# 0.0.22: an unowned gate has no owner scope (run 20: "Invalid context switch [owner]").
				limit = { OR = { NOT = { exists = owner } NOT = { owner = { is_same_value = root } } } }
				set_owner = root
				log = "AHW SPIKE [Root.GetName] now owns the gate in [Prev.GetName]"
			}
			if = {
				limit = { has_star_flag = ahw_sealed }
				ahw_spike_seal_pair_for = { COUNTRY = root }
			}
			if = {
				limit = {
					NOT = { has_star_flag = ahw_studied }
					NOT = { has_star_flag = ahw_study_offered }
				}
				set_star_flag = ahw_study_offered
				enable_special_project = { name = AHW_SPIKE_STUDY_PROJECT location = this owner = root }
				log = "AHW SPIKE study of the entry point in [This.GetName] offered to [Root.GetName]"
			}
			else_if = {
				limit = {
					has_star_flag = ahw_studied
					has_star_flag = ahw_broken
					NOT = { has_star_flag = ahw_repair_offered }
					root = { has_technology = ahw_tech_spike }
				}
				set_star_flag = ahw_repair_offered
				save_event_target_as = ahw_repair_portal
				root = { country_event = { id = ahw_spike.21 } }
				log = "AHW SPIKE repair of the entry point in [This.GetName] offered to [Root.GetName]"
			}
		}
	}
}

# The repair offer: pay alloys, then a construction ship repairs the gate. "Not now" offers it
# again in a year. Special projects cannot charge resources themselves.
country_event = {
	id = ahw_spike.21
	title = ahw_spike.21.name
	desc = ahw_spike.21.desc
	is_triggered_only = yes
	option = {
		name = ahw_spike.21.a
		allow = { has_resource = { type = alloys amount >= 500 } }
		add_resource = { alloys = -500 }
		event_target:ahw_repair_portal = {
			enable_special_project = { name = AHW_SPIKE_REPAIR_PROJECT location = this owner = root }
		}
		ai_chance = { factor = 10 }
	}
	option = {
		name = ahw_spike.21.b
		event_target:ahw_repair_portal = {
			remove_star_flag = ahw_repair_offered
			set_timed_star_flag = { flag = ahw_repair_offered days = 360 }
		}
		ai_chance = { factor = 1 }
	}
}

# 0.0.31 (user, 2026-11-06): every regular empire without a portal gets one - empires born later
# (rebels, secessions, released subjects) and those for which no anchor was found before. Monthly
# (on_monthly_pulse_country). Run 30: a spawn in play freezes the game ~10 s on a huge galaxy
# (~2.5 s medium), accepted; the spawn, the lane and the gates share one batch, opened only when an
# anchor exists, so an empire still without one costs no freeze.
country_event = {
	id = ahw_spike.30
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_country_type = default
		has_global_flag = ahw_spike_done
		NOT = { has_country_flag = ahw_has_portal }
		exists = capital_star
		ahw_spike_pair_left = yes
	}

	immediate = {
		capital_star = { solar_system = { save_global_event_target_as = ahw_cap_tmp } }
		save_global_event_target_as = ahw_country_tmp
		if = {
			limit = {
				any_system = {
					OR = {
						ahw_spike_portal_anchor_ok = yes
						ahw_spike_portal_anchor_own_ok = yes
					}
					ahw_spike_within_3_jumps_of_capital = yes
				}
			}
			set_spawn_system_batch = begin
			ahw_spike_build_portal = yes
			set_spawn_system_batch = end
			if = {
				limit = { has_country_flag = ahw_has_portal }
				event_target:ahw_portal_tmp = { ahw_spike_pick_centre = yes ahw_spike_add_debris = yes }
				log = "AHW SPIKE late portal for [This.GetName] (capital [ahw_cap_tmp.GetName]) on [GetDate]: [ahw_portal_tmp.GetName], off [ahw_x_tmp.GetName], exit in [ahw_node_tmp.GetName]"
			}
		}
	}
}
