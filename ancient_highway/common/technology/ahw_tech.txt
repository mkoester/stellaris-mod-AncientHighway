# Two techs (design 2026-11-05), researched in parallel. Researchable once the
# country has studied an entry point (special project AHW_STUDY_PROJECT -> country flag
# ahw_highway_studied; add_research_option puts them among the options at once, as vanilla's
# gateway study does) or got the head start (ahw.20). Tier 1: its previously_unlocked is 0,
# so no tier prerequisite (common/technology/tier/00_tier.txt). Run 3: research_technology refuses a
# tech whose potential fails; `event ahw_debug.8` grants both anyway (give_technology).
# Icons: vanilla's gateway techs', referenced by name (no copy of game art).

# Physics: the gates' theory - use a gate, seal one.
ahw_tech_gate_theory = {
	icon = "tech_gateway_activation"
	cost = @tier2cost2
	area = physics
	tier = 1
	category = { particles }
	is_rare = yes
	is_reverse_engineerable = no
	potential = {
		has_country_flag = ahw_highway_studied
	}
	weight = 100
	# run 43: without an ai_weight no AI empire researched a gate tech (vanilla's priorities are
	# 1.25-2, must-haves 100). A flat 3 (2026-11-18) gave the first tech 4-6 years later and no
	# empire both in 26 (the user's campaign). Since 2026-11-20 (user) the weight grows with what
	# the grants have paid off (the country variable of this tech's grants: study, inspections,
	# travel techs, others' knowledge), so the AI does not buy it early at full price: x1 below
	# 30 %, x3 from 30 %, x6 from 50 %, x12 from 70 % (cap 80 %). x4 with the other gate tech: one
	# alone is useless (2226: seven empires with one each). Need stays on top: x3 with a friend
	# out of reach, x5 more if also threatened. The engine adds x1.5 for rare techs.
	ai_weight = {
		factor = 1
		modifier = { factor = 3 check_variable = { which = ahw_got_theory value >= 0.3 } }
		modifier = { factor = 2 check_variable = { which = ahw_got_theory value >= 0.5 } }
		modifier = { factor = 2 check_variable = { which = ahw_got_theory value >= 0.7 } }
		modifier = { factor = 4 has_technology = ahw_tech_gate_construction }
		modifier = { factor = 3 ahw_far_friend = yes }
		modifier = { factor = 5 ahw_threatened = yes ahw_far_friend = yes }
	}
}
