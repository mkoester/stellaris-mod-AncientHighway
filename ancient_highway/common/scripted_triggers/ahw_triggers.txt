# this = system: may the portal hang off this system? Not the capital
# (event_target:ahw_cap_tmp), not next to it, not artificial, not owned (game start: only home
# systems are). Used with ahw_within_3_jumps_of_capital: 2 or 3 jumps out.
ahw_portal_anchor_ok = {
	NOT = { is_same_value = event_target:ahw_cap_tmp }
	NOT = { has_hyperlane_to = event_target:ahw_cap_tmp }
	NOT = { has_star_flag = ahw_unclaimable }
	NOT = { exists = owner }
	# Run 12: two empires shared the dead end Tiblic; the second portal's lane arrived beside the
	# first's (margin -0.55). One portal per anchor.
	NOT = { any_neighbor_system = { has_star_flag = ahw_portal } }
}

# 0.0.31 (user, 2026-11-06): in a running game most systems near a capital are owned; the anchor
# may then be one the empire (event_target:ahw_country_tmp) owns itself, never another's.
ahw_portal_anchor_own_ok = {
	NOT = { is_same_value = event_target:ahw_cap_tmp }
	NOT = { has_hyperlane_to = event_target:ahw_cap_tmp }
	NOT = { has_star_flag = ahw_unclaimable }
	exists = owner
	owner = { is_same_value = event_target:ahw_country_tmp }
	NOT = { any_neighbor_system = { has_star_flag = ahw_portal } }
}

# 0.0.41, this = country: gets a portal - a regular empire, not a nomad (design: nomads get no
# portal; run 38's nomadic Murmies had one). is_nomadic is the engine's country code flag, set
# before on_game_start (vanilla's nomads.20 tests it from on_game_start_country); an empire that
# settles counts from then on, one that embarks keeps the portal it has.
ahw_gets_portal = {
	is_country_type = default
	is_nomadic = no
}

# 0.0.31: a generated pair is left for one more portal (ahw_pairs counts the used ones, ahw_pairs_max
# is tools/gen.py's PAIRS; both on ring node 1). No scope needed.
ahw_pair_left = {
	event_target:ahw_ring_1 = { check_variable = { which = ahw_pairs value < ahw_pairs_max } }
}

# Run 11: the random walk saw one path only and missed 3-jump candidates; ahw.10 now
# searches every system. this = system: 1 to 3 jumps from the capital (with
# ahw_portal_anchor_ok, which rules out the capital and its neighbours: 2 to 3 jumps).
ahw_within_3_jumps_of_capital = {
	OR = {
		has_hyperlane_to = event_target:ahw_cap_tmp
		any_neighbor_system = { has_hyperlane_to = event_target:ahw_cap_tmp }
		any_neighbor_system = { any_neighbor_system = { has_hyperlane_to = event_target:ahw_cap_tmp } }
	}
}

# this = system: already holds a gateway, wormhole or other bypass (user: de-prioritise these).
ahw_has_bypass = {
	any_bypass_in_system = { always = yes }
}

# 0.0.24 (user, 2026-11-05: "the whole idea is that empires can still reach their allies - every
# type of friend must still be allowed through (esp. empires fighting on the same side of a war)").
# this = country: the sealer of pair $N$ (event_target:ahw_sealer_$N$) or a friend of it. A pair
# sealed against everyone (ahw_debug.25) has no sealer, so nobody passes.
ahw_friend_of_sealer = {
	exists = event_target:ahw_sealer_$N$
	OR = {
		is_same_value = event_target:ahw_sealer_$N$
		# one federation
		is_in_federation_with = event_target:ahw_sealer_$N$
		# overlord and subject, either way, and two subjects of one overlord
		is_overlord_to = event_target:ahw_sealer_$N$
		event_target:ahw_sealer_$N$ = { is_overlord_to = prev }
		AND = {
			is_subject = yes
			event_target:ahw_sealer_$N$ = { is_subject = yes }
			overlord = { is_overlord_to = event_target:ahw_sealer_$N$ }
		}
		# pacts and guarantees, either way
		has_defensive_pact = event_target:ahw_sealer_$N$
		has_non_aggression_pact = event_target:ahw_sealer_$N$
		has_commercial_pact = event_target:ahw_sealer_$N$
		has_research_agreement = event_target:ahw_sealer_$N$
		has_migration_access = event_target:ahw_sealer_$N$
		event_target:ahw_sealer_$N$ = { has_migration_access = prev }
		is_guaranteeing = event_target:ahw_sealer_$N$
		event_target:ahw_sealer_$N$ = { is_guaranteeing = prev }
		# at peace with each other, borders open both ways (user, 2026-11-05). Communications first:
		# unmet empires are no friends, whatever their border setting reads.
		AND = {
			has_communications = event_target:ahw_sealer_$N$
			NOT = { is_at_war_with = event_target:ahw_sealer_$N$ }
			NOT = { has_closed_borders = event_target:ahw_sealer_$N$ }
			event_target:ahw_sealer_$N$ = { NOT = { has_closed_borders = prev } }
		}
		# nomads, unless the sealer banned them (user, 2026-11-05): at war with it, or expelled -
		# vanilla's timed flag after an Expel Nomads war (10 years, nomads_effects.txt; game rule
		# is_expelled_from)
		AND = {
			is_nomadic = yes
			NOT = { is_at_war_with = event_target:ahw_sealer_$N$ }
			NOT = { has_country_flag = expelled_from_@event_target:ahw_sealer_$N$ }
		}
		# the same side of any war
		any_war = {
			OR = {
				AND = {
					is_war_participant = { who = prev side = attackers }
					is_war_participant = { who = event_target:ahw_sealer_$N$ side = attackers }
				}
				AND = {
					is_war_participant = { who = prev side = defenders }
					is_war_participant = { who = event_target:ahw_sealer_$N$ side = defenders }
				}
			}
		}
	}
}

# this = a portal: WHO has seen a centre of its kind (ahw_see_kind).
ahw_kind_seen = {
	OR = {
		AND = { is_star_class = ahw_sc_black_hole $WHO$ = { has_country_flag = ahw_kind_black_hole } }
		AND = { is_star_class = ahw_sc_pulsar $WHO$ = { has_country_flag = ahw_kind_pulsar } }
		AND = { is_star_class = ahw_sc_toxoid_star $WHO$ = { has_country_flag = ahw_kind_toxoid } }
		AND = { is_star_class = ahw_sc_neutron_star $WHO$ = { has_country_flag = ahw_kind_neutron } }
	}
}

# 0.1.0, this = system: an anchor - next to a portal whose gate is still broken (the contract).
ahw_anchor_of_broken_portal = {
	any_neighbor_system = {
		has_star_flag = ahw_portal
		has_star_flag = ahw_broken
	}
}

# 0.1.0, this = country: a close friend of $WHO$ - for the repair contract on a friend's gate
# (design 2026-11-05): stricter than a seal's friend (ahw_friend_of_sealer) - federation,
# overlord/subject either way, fellow subjects, defensive pact, guarantee either way, the same
# side of a war; not a non-aggression pact, a migration treaty or open borders.
ahw_close_friend = {
	OR = {
		is_in_federation_with = $WHO$
		is_overlord_to = $WHO$
		$WHO$ = { is_overlord_to = prev }
		AND = {
			is_subject = yes
			$WHO$ = { is_subject = yes }
			overlord = { is_overlord_to = $WHO$ }
		}
		has_defensive_pact = $WHO$
		is_guaranteeing = $WHO$
		$WHO$ = { is_guaranteeing = prev }
		any_war = {
			OR = {
				AND = {
					is_war_participant = { who = prev side = attackers }
					is_war_participant = { who = $WHO$ side = attackers }
				}
				AND = {
					is_war_participant = { who = prev side = defenders }
					is_war_participant = { who = $WHO$ side = defenders }
				}
			}
		}
	}
}

# 0.1.1, this = country: owns an anchor whose portal's gate is still broken.
ahw_owns_broken_anchor = {
	any_system_within_border = { ahw_anchor_of_broken_portal = yes }
}

# 0.1.1, this = country: in breach of "Unrepaired Gates" - it could repair (both techs) a gate on
# its own anchor and has not; vanilla's Constitutional Immunity exempts.
ahw_gate_breach = {
	NOT = { has_country_flag = constitutional_immunity }
	has_technology = ahw_tech_gate_theory
	has_technology = ahw_tech_gate_construction
	ahw_owns_broken_anchor = yes
}

# 0.1.1, this = country: owns an anchor whose gate is sealed.
ahw_owns_sealed_gate = {
	any_system_within_border = {
		any_neighbor_system = { has_star_flag = ahw_portal has_star_flag = ahw_sealed }
	}
}

# 0.1.1, this = country: owns an anchor whose gate is repaired.
ahw_owns_repaired_gate = {
	any_system_within_border = {
		any_neighbor_system = { has_star_flag = ahw_portal NOT = { has_star_flag = ahw_broken } }
	}
}

# 1.0.1, this = country: owns a repaired gate and can travel through it - Gate Theory, the
# bypasses' prerequisite (common/bypass/ahw_bypasses.txt). A gate a nomad repaired for an empire
# without it is not enough (user, 2026-11-23). Who may propose a highway resolution.
ahw_can_use_own_gate = {
	ahw_owns_repaired_gate = yes
	has_technology = ahw_tech_gate_theory
}

# 0.1.1, this = country (AI weights): a friend out of reach - federation, defensive pact, our
# overlord or our guarantor, and not a neighbour ("not a neighbour" stands in for "out of reach":
# no reachability trigger exists). A subset of ahw_close_friend, the symmetric relations only.
ahw_far_friend = {
	any_country = {
		is_country_type = default
		NOT = { is_neighbor_of = prev }
		OR = {
			is_in_federation_with = prev
			has_defensive_pact = prev
			is_overlord_to = prev
			is_guaranteeing = prev
		}
	}
}

# 0.1.1, this = country (AI weights): threatened - a neighbour with a superior fleet, or an empire
# that counts us its rival.
ahw_threatened = {
	OR = {
		any_neighbor_country = { relative_power = { who = prev category = fleet value = superior } }
		any_country = { is_rival = prev }
	}
}

# 2026-11-19, this = a country: a foreign portal's broken gate should feed Gate Construction now -
# Construction is open and was granted less than Theory so far (Theory wins a tie), or Theory is
# closed (capped or researched; ahw_cap_check). ahw.60 offers the centre study otherwise.
ahw_construction_behind = {
	NOT = { has_country_flag = ahw_closed_construction }
	OR = {
		has_country_flag = ahw_closed_theory
		AND = {
			is_variable_set = ahw_got_theory
			NOT = { is_variable_set = ahw_got_construction }
		}
		AND = {
			is_variable_set = ahw_got_theory
			is_variable_set = ahw_got_construction
			check_variable = { which = ahw_got_construction value < ahw_got_theory }
		}
	}
}
