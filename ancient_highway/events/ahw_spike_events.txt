# SPIKE (Phase 0, 0.0.6): can the Ancient Highway be built from the engine's parts?
# ahw_spike.1 runs by itself at game start (common/on_actions/ahw_spike_on_actions.txt).
# Console, in a throwaway game, with nothing selected (Esc) - see AGENTS.md, "Status":
#   event ahw_spike.2   report the state to game.log
#   event ahw_spike.3   lock entry point 1 (our capital) for us (360 days)
#   event ahw_spike.4   close entry point 1 (remove its "open" flag)
#   event ahw_spike.5   reopen it (set the flag again)
#   event ahw_spike.6   tear down its gate; report now and a day later
#   event ahw_spike.8   grant ahw_tech_spike (or research_technology ahw_tech_spike)
# Map changes reach route-finding on the next daily tick: let a day pass before ordering routes.
namespace = ahw_spike

country_event = {
	id = ahw_spike.1
	title = ahw_spike.1.name
	desc = ahw_spike.1.desc
	is_triggered_only = yes

	trigger = {
		is_ai = no
		NOT = { has_global_flag = ahw_spike_done }
	}

	immediate = {
		set_global_flag = ahw_spike_done
		log = "AHW SPIKE ===== ahw_spike.1: building the ring at game start ====="

		# How empty is the core (fraction of the galaxy radius)?
		ahw_spike_core_band = { FRACTION = 0.05 }
		ahw_spike_core_band = { FRACTION = 0.1 }
		ahw_spike_core_band = { FRACTION = 0.15 }
		ahw_spike_core_band = { FRACTION = 0.2 }
		ahw_spike_core_band = { FRACTION = 0.25 }

		# The ring goes next to our capital so ships reach it fast; placing it around the core is
		# a separate spike. A: our capital system, B: a neighbour, C: another neighbour (random
		# each game - the event window and the log name them).
		capital_star = { solar_system = { save_global_event_target_as = ahw_anchor } }
		event_target:ahw_anchor = {
			random_neighbor_system = {
				save_global_event_target_as = ahw_anchor_b
			}
			random_neighbor_system = {
				limit = { NOT = { is_same_value = event_target:ahw_anchor_b } }
				save_global_event_target_as = ahw_anchor_c
			}
		}

		# Six ring systems, each 40 from the previous one, turning 60 degrees each step
		# (run 2: the angle is an absolute map direction).
		ahw_spike_spawn_ring_step = { FROM = ahw_anchor NAME = ahw_ring_1 DIST = 30 ANGLE = 0 }
		ahw_spike_spawn_ring_step = { FROM = ahw_ring_1 NAME = ahw_ring_2 DIST = 40 ANGLE = 60 }
		ahw_spike_spawn_ring_step = { FROM = ahw_ring_2 NAME = ahw_ring_3 DIST = 40 ANGLE = 120 }
		ahw_spike_spawn_ring_step = { FROM = ahw_ring_3 NAME = ahw_ring_4 DIST = 40 ANGLE = 180 }
		ahw_spike_spawn_ring_step = { FROM = ahw_ring_4 NAME = ahw_ring_5 DIST = 40 ANGLE = 240 }
		ahw_spike_spawn_ring_step = { FROM = ahw_ring_5 NAME = ahw_ring_6 DIST = 40 ANGLE = 300 }

		# Ring hyperlanes, closing the loop. No hyperlane to the regular galaxy.
		add_hyperlane = { from = event_target:ahw_ring_1 to = event_target:ahw_ring_2 }
		add_hyperlane = { from = event_target:ahw_ring_2 to = event_target:ahw_ring_3 }
		add_hyperlane = { from = event_target:ahw_ring_3 to = event_target:ahw_ring_4 }
		add_hyperlane = { from = event_target:ahw_ring_4 to = event_target:ahw_ring_5 }
		add_hyperlane = { from = event_target:ahw_ring_5 to = event_target:ahw_ring_6 }
		add_hyperlane = { from = event_target:ahw_ring_6 to = event_target:ahw_ring_1 }

		# Entry points, 1:1, one bypass type each; every flag before any gate.
		#   1: A <-> exit 1 in ring 1 (orbit 300, angle 0)    repaired
		#   2: B <-> exit 2 in ring 1 (orbit 300, angle 120)  repaired
		#   3: C <-> exit 3 in ring 2 (orbit 300, angle 0)    broken (no "open" flag)
		event_target:ahw_anchor = { set_star_flag = ahw_entry_open_1 }
		event_target:ahw_anchor_b = { set_star_flag = ahw_entry_open_2 }
		event_target:ahw_ring_1 = {
			set_star_flag = ahw_exit_1
			set_star_flag = ahw_exit_2
		}
		event_target:ahw_ring_2 = { set_star_flag = ahw_exit_3 }

		event_target:ahw_anchor = {
			spawn_megastructure = {
				type = ahw_spike_entry_gate_1
				owner = root # our capital system: shows in our outliner
				orbit_distance = 350
				orbit_angle = 45
				init_effect = { save_global_event_target_as = ahw_entry_gate_a }
			}
		}
		event_target:ahw_anchor_b = {
			spawn_megastructure = {
				type = ahw_spike_entry_gate_2
				orbit_distance = 350
				orbit_angle = 45
			}
		}
		event_target:ahw_anchor_c = {
			spawn_megastructure = {
				type = ahw_spike_entry_gate_3
				orbit_distance = 350
				orbit_angle = 45
			}
		}
		event_target:ahw_ring_1 = {
			spawn_megastructure = {
				type = ahw_spike_entry_exit_1
				orbit_distance = 300
				orbit_angle = 0
			}
			spawn_megastructure = {
				type = ahw_spike_entry_exit_2
				orbit_distance = 300
				orbit_angle = 120
			}
		}
		event_target:ahw_ring_2 = {
			spawn_megastructure = {
				type = ahw_spike_entry_exit_3
				orbit_distance = 300
				orbit_angle = 0
			}
		}

		# Shortcut ring_1 <-> ring_4: a gate pair on its own scripted connection (run 5: as a
		# natural wormhole it set off vanilla's first-wormhole story).
		event_target:ahw_ring_1 = { set_star_flag = ahw_shortcut_1_nexus }
		event_target:ahw_ring_4 = { set_star_flag = ahw_shortcut_1_node }
		event_target:ahw_ring_1 = {
			spawn_megastructure = {
				type = ahw_spike_shortcut_1_nexus
				orbit_distance = 300
				orbit_angle = 240
			}
		}
		event_target:ahw_ring_4 = {
			spawn_megastructure = {
				type = ahw_spike_shortcut_1_node
				orbit_distance = 300
				orbit_angle = 60
			}
		}

		# Before use (run 4): the country must have seen the types and know the ring systems.
		# In the mod this is what studying an entry point would teach.
		add_seen_bypass_type = ahw_spike_entry_1
		add_seen_bypass_type = ahw_spike_entry_2
		add_seen_bypass_type = ahw_spike_entry_3
		add_seen_bypass_type = ahw_spike_shortcut_1
		every_system = {
			limit = { has_star_flag = ahw_ring }
			root = { set_visited = prev }
		}

		ahw_spike_report = { WHEN = at_game_start }
	}

	option = { name = ahw_spike.ok }
}

country_event = {
	id = ahw_spike.2
	title = ahw_spike.2.name
	desc = ahw_spike.2.desc
	is_triggered_only = yes
	immediate = { ahw_spike_report = { WHEN = on_request } }
	option = { name = ahw_spike.ok }
}

country_event = {
	id = ahw_spike.3
	title = ahw_spike.3.name
	desc = ahw_spike.3.desc
	is_triggered_only = yes
	immediate = {
		event_target:ahw_anchor = {
			random_bypass_in_system = {
				limit = { is_bypass_type = ahw_spike_entry_1 }
				lock_bypass = {
					country = root
					duration = 360
				}
			}
		}
		ahw_spike_report = { WHEN = after_lock }
	}
	option = { name = ahw_spike.ok }
}

country_event = {
	id = ahw_spike.4
	title = ahw_spike.4.name
	desc = ahw_spike.4.desc
	is_triggered_only = yes
	immediate = {
		event_target:ahw_anchor = { remove_star_flag = ahw_entry_open_1 }
		ahw_spike_report = { WHEN = after_closing_A }
	}
	option = { name = ahw_spike.ok }
}

country_event = {

	title = ahw_spike.5.name
	desc = ahw_spike.5.desc
	is_triggered_only = yes
	immediate = {
		event_target:ahw_anchor = { set_star_flag = ahw_entry_open_1 }
		ahw_spike_report = { WHEN = after_reopening_A }
	}
	option = { name = ahw_spike.ok }
}

country_event = {
	id = ahw_spike.6
	title = ahw_spike.6.name
	desc = ahw_spike.6.desc
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { exists = event_target:ahw_entry_gate_a }
			remove_megastructure = event_target:ahw_entry_gate_a
		}
		ahw_spike_report = { WHEN = right_after_teardown_A }
		country_event = { id = ahw_spike.7 days = 1 }
	}
	option = { name = ahw_spike.ok }
}

country_event = {
	id = ahw_spike.7
	hide_window = yes
	is_triggered_only = yes
	immediate = { ahw_spike_report = { WHEN = one_day_after_teardown_A } }
}

# Grants the entry-point tech (fallback for the console's research_technology).
country_event = {
	id = ahw_spike.8
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		give_technology = { tech = ahw_tech_spike message = no }
		ahw_spike_report = { WHEN = after_granting_the_tech }
	}
}
