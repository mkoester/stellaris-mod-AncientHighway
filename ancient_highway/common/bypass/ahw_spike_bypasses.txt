# SPIKE (Phase 0). Field meanings: the header of the vanilla common/bypass/00_bypasses.txt.

# Entry points of ring system 1 (the real mod: one such type per ring system, generated).
# Run 2: link_wormholes refuses megastructure bypasses, so entry points use a scripted connection,
# as the L-gates do: the nexus is the ring system, nodes are regular systems whose entry point is
# repaired. Nodes reach only the nexus, never each other.
# Q: does removing ahw_entry_open_1 from a node close it (Shroud tunnels use a dynamic node
# trigger)? Is it usable only with ahw_tech_spike? Does lock_bypass work on it?
ahw_spike_entry_1 = {
	name = AHW_SPIKE_ENTRY
	is_pathfind = yes
	uses_action = no
	requires_exploration = no
	always_active = yes
	windup_time = 0
	winddown_time = 0
	ftl_multiplier = 0
	extends_sensors = no
	icon_frame = 25
	remember_instances = no
	animation_time = 1
	galactic_map_tooltip_header = AHW_SPIKE_ENTRY
	galactic_map_tooltip_desc = AHW_SPIKE_ENTRY_DESC
	connection_type = scripted_connection
	ftl_ship_effect = "gateway_ftl_ship_effect_entity"
	ftl_bypass_effect = "gateway_ftl_bypass_effect_entity"
	can_bypass_be_locked = yes

	scripted_connection = {
		node = { # system scope
			has_star_flag = ahw_entry_open_1
		}
		nexus = { # system scope
			has_star_flag = ahw_nexus_1
		}
	}

	prerequisites = {
		ahw_tech_spike
	}

	on_action = "on_entering_wormhole"
}
