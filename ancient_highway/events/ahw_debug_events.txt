# same effects as the real stage it skips.
namespace = ahw_debug

# An outpost on our anchor, so we own it (the anchor's owner studies, repairs and locks).
country_event = {
	id = ahw_debug.6
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		event_target:ahw_player_anchor = {
			if = {
				limit = { NOT = { exists = owner } }
				create_starbase = { size = starbase_outpost owner = root }
			}
			log = "AHW anchor [This.GetName]: outpost created for [Root.GetName]"
		}
	}
}

			}

# Test (0.0.39): our three test fleets at the anchor of a foreign portal - one whose centre kind we
# have not seen, if there is one - so that one jump takes them into it.
country_event = {
	id = ahw_debug.46
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				any_system = {
					has_star_flag = ahw_portal
					NOT = { is_same_value = event_target:ahw_player_portal }
					NOT = { ahw_kind_seen = { WHO = root } }
				}
			}
			random_system = {
				limit = {
					has_star_flag = ahw_portal
					NOT = { is_same_value = event_target:ahw_player_portal }
					NOT = { ahw_kind_seen = { WHO = root } }
				}
				save_event_target_as = ahw_foreign_portal
			}
		}
		else = {
			random_system = {
				limit = { has_star_flag = ahw_portal NOT = { is_same_value = event_target:ahw_player_portal } }
				save_event_target_as = ahw_foreign_portal
			}
		}
		event_target:ahw_foreign_portal = {
			random_neighbor_system = { save_event_target_as = ahw_spawn_here }
			log = "AHW test ships next to the foreign portal [This.GetName]"
		}
		ahw_spawn_test_ships = yes
	}
}

# 0.1.0: a nomad ready to repair our gate - in contact with us, both gate techs, its arkship at
# our anchor's star (one jump from our portal). For the nomad's own repair (the upgrade) and the
# contract; game.log names it.
country_event = {
	id = ahw_debug.73
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		random_country = {
			limit = { is_country_type = default is_nomadic = yes }
			save_event_target_as = ahw_test_nomad
		}
		if = {
			limit = { exists = event_target:ahw_test_nomad }
			event_target:ahw_player_anchor = { star = { save_event_target_as = ahw_test_star } }
			event_target:ahw_test_nomad = {
				establish_communications_no_message = root
				# run 40: our techs are only valid for a country that has studied a gate (their
				# potential) - "Attempting to give invalid technology" otherwise
				set_country_flag = ahw_highway_studied
				ahw_learn_types = yes
				give_technology = { tech = ahw_tech_gate_theory message = no }
				give_technology = { tech = ahw_tech_gate_construction message = no }
				random_owned_fleet = {
					limit = { is_arkship_fleet = yes }
					set_location = { target = event_target:ahw_test_star distance = 0 angle = random }
				}
				log = "AHW test nomad [This.GetName]: in contact with [Root.GetName], both techs, arkship at [ahw_test_star.GetName]"
			}
		}
		else = {
			log = "AHW test nomad: there is no nomadic empire in this game"
		}
	}
}

# 0.1.0 (run 43: the AI took 5-12 years to claim its anchor and study): every AI empire's portal
# studied now - an outpost on its anchor if nobody owns it, then the study's real result
# (ahw_mark_studied). To watch the AI research the gate techs without waiting decades.
country_event = {
	id = ahw_debug.74
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_country = {

				limit = { has_star_flag = ahw_portal_for@prev }
				random_neighbor_system = {
					limit = { NOT = { exists = owner } }

				}
				if = {
					limit = { NOT = { has_star_flag = ahw_studied } }
					set_star_flag = ahw_study_offered
					ahw_mark_studied = yes
				}
				log = "AHW test: [ahw_studier.GetName] studied its portal [This.GetName]"
			}
		}
	}
}

# 0.1.1 (run 44: test what the AI does with a gate without waiting decades): every AI empire's
# portal studied, both techs, and every other one repaired - an outpost on its anchor first if
# nobody owns it. The broken half shows whether the AI (or a nomad) repairs by itself, the
# repaired half whether it travels. The real stages' effects, as ahw_debug.40.
country_event = {
	id = ahw_debug.75
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_variable = { which = ahw_n_alt value = 0 }
		every_country = {
			limit = { is_ai = yes is_country_type = default has_country_flag = ahw_has_portal }
			save_event_target_as = ahw_studier
			save_event_target_as = ahw_repairer
			root = { change_variable = { which = ahw_n_alt value = 1 } }
			random_system = {
				limit = { has_star_flag = ahw_portal_for@prev }
				random_neighbor_system = {
					limit = { NOT = { exists = owner } }
					create_starbase = { size = starbase_outpost owner = event_target:ahw_studier }
				}
				if = {
					limit = { NOT = { has_star_flag = ahw_studied } }
					set_star_flag = ahw_study_offered
					ahw_mark_studied = yes
				}
				if = {
					limit = { root = { check_variable = { which = ahw_n_alt value = 2 } } }
					ahw_repair_gate = yes
					log = "AHW test: [ahw_studier.GetName] studied its portal [This.GetName], both techs, REPAIRED"
				}
				else = {
					log = "AHW test: [ahw_studier.GetName] studied its portal [This.GetName], both techs, left broken"
				}
			}
			give_technology = { tech = ahw_tech_gate_theory message = no }
			give_technology = { tech = ahw_tech_gate_construction message = no }
			root = {
				if = {
					limit = { check_variable = { which = ahw_n_alt value = 2 } }
					set_variable = { which = ahw_n_alt value = 0 }
				}
			}
		}
	}
}

# 0.1.1: a friend beyond our borders at war (ally under attack, ahw.95) - an AI empire that is no
# neighbour of ours becomes our subject, and one of its neighbours declares war on it. The
# message comes on the next 1st, if we have studied a gate.
country_event = {
	id = ahw_debug.78
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		random_country = {
			limit = {
				is_ai = yes
				is_country_type = default
				is_nomadic = no
				is_subject = no
				NOT = { is_neighbor_of = root }
				any_neighbor_country = { is_country_type = default is_nomadic = no NOT = { is_same_value = root } is_subject = no }
			}
			save_global_event_target_as = ahw_test_ally
			set_subject_of = { who = root preset = preset_vassal }
			random_neighbor_country = {
				limit = { is_country_type = default is_nomadic = no NOT = { is_same_value = root } is_subject = no }
				declare_war = { target = event_target:ahw_test_ally attacker_war_goal = "wg_humiliation" }
				log = "AHW test: [ahw_test_ally.GetName], now our subject, attacked by [This.GetName]"
			}
		}
		if = {
			limit = { NOT = { has_country_flag = ahw_highway_studied } }
			log = "AHW test: we have studied no gate - ahw.95 stays silent until we do (ahw_debug.45)"
		}
	}
}
