# Two techs (design 2026-11-05), researched in parallel. Researchable once the
# country has studied an entry point (special project AHW_STUDY_PROJECT -> country flag
# ahw_highway_studied; add_research_option puts them among the options at once, as vanilla's
# gateway study does) or got the head start (ahw.20). Tier 1: its previously_unlocked is 0,
# so no tier prerequisite (common/technology/tier/00_tier.txt). Run 3: research_technology refuses a
# tech whose potential fails; `event ahw_debug.8` grants both anyway (give_technology).
# Icons: vanilla's gateway techs', referenced by name (no copy of game art).

# Physics: the gates' theory - use a gate, seal one.
ahw_tech_gate_theory = {
	icon = "tech_gateway_activation"
	cost = @tier2cost2
	area = physics
	tier = 1
	category = { particles }
	is_rare = yes
	is_reverse_engineerable = no
	potential = {
		has_country_flag = ahw_highway_studied
	}
	weight = 100
	# run 43: without an ai_weight no AI empire researched a gate tech in 13 years after its
	# study (seven had the option). Vanilla's priorities are 1.25-2, must-haves 100; 3 is a
	# clear preference, not a rush (tunable). The base stays low on purpose (user, 2026-11-18:
	# no empire needs the highway at year ~13, more likely around year 30-40); need raises it
	# (design.md, "Threat-based AI weights"). The engine adds x1.5 for a rare tech
	# (AI_TECHNOLOGY_RARE_WEIGHT_MULT): 4.5 studied only, 13.5 with a friend out of reach, 67.5
	# also threatened. Not "owns a broken anchor": only an anchor's owner studies, so that would
	# be true for nearly every empire with the option - a raised base. The user's campaign
	# (2026-11-18): 3 alone picked by one of 13 empires in up to 12 years.
	ai_weight = {
		factor = 3
		modifier = { factor = 3 ahw_far_friend = yes }
		modifier = { factor = 5 ahw_threatened = yes ahw_far_friend = yes }
	}
}
