# The Ancient Highway's story (2026-11-19, docs/lore.md "How players learn it"): short pop-ups
# with one button, for human players only (AI countries see nothing and get no flags).
# - Three core chapters every player sees, whether they explore or research straight away:
#   ahw_story.1 the study (what it is, what now), .2 the first gate tech (by whom), .3 the first
#   ring node entered (why). Each once (country flag ahw_story_<key>).
# - Fragments: ahw_story.10 picks one not yet seen, at random, from a pool of 12 (.11-.22) -
#   on the first entry into each foreign portal and each reach after the first (ahw.60,
#   ahw_story_entered). Each readable on its own, so the order does not matter.
# Pictures are vanilla's, referenced by name.
namespace = ahw_story

# The study done, or the head start (ahw_country_learns): what the gate is, and what to do now.
country_event = {
	id = ahw_story.1
	title = ahw_story.1.name
	desc = ahw_story.1.desc
	picture = GFX_evt_alien_ruins
	is_triggered_only = yes
	trigger = {
		is_ai = no
		NOT = { has_country_flag = ahw_story_door }
	}
	immediate = { set_country_flag = ahw_story_door }
	option = { name = ahw_story.1.a }
}

# counter stops the call before random_list would find no entry).

	id = ahw_story.10
	hide_window = yes
	is_triggered_only = yes

		is_ai = no

		}

		if = {
			limit = { NOT = { is_variable_set = ahw_n_frags } }

		random_list = {
